package main

import (
	"reflect"

	"github.com/labstack/echo/v4"
)

func main(){
	e:= echo.New()

	e.GET("/", func(c echo.Context) error {
		return c.String(200, "Hello, World!")
	})
	e.GET("/users", GetUsers)
	e.GET("/user/:id", GetUserByID)
	e.POST("/user",CreateUser)
	e.Logger.Fatal(e.Start(":8080"))
}

type User struct {
	ID   string `json:"id"`
	Name string `json:"name"`
	Age  int    `json:"age"`
}

type Response struct{
	ResponseCode int   `json:"response_code"`
	ResponseMessage string `json:"response_message"`
	Data interface{} `json:"data,omitempty"`
}

var data = []User{
	{ID: "1", Name: "John Doe 1", Age: 30},
	{ID: "2", Name: "John Doe 2", Age: 30},
	{ID: "3", Name: "John Doe 3", Age: 30},
}

func GetUsers(c echo.Context) error {
	return c.JSON(200, data)
}

func GetUserByID(c echo.Context) error {
	// Dummy implementation
	id := c.Param("id")

	dataMap := make(map[string]User)
	for _, user := range data {
		dataMap[user.ID] = user
	}

	if user, exists := dataMap[id]; exists {
		return c.JSON(200, user)
	}	
	return c.JSON(404, map[string]string{"message": "User not found"})	
}

func CreateUser(c echo.Context) error {
	var requestUser User
	var Response Response

	if err := c.Bind(&requestUser);err != nil {
		Response.ResponseCode = 400
		Response.ResponseMessage = "Invalid request"
		return c.JSON(400, Response)
	}
	if reflect.DeepEqual(requestUser, User{}) {
		Response.ResponseCode = 400
		Response.ResponseMessage = "Request body cannot be empty"
		return c.JSON(400, Response)
	}

	data = append(data, requestUser)

	Response.ResponseCode = 201
	Response.ResponseMessage = "User created successfully"
	Response.Data = requestUser
	return c.JSON(201, Response)
}
